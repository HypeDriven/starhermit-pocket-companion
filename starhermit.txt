# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.19.2
engine.minor=20
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The action tray's slots act in the tray's order (decorate opens the
# decorations); the tray, Mote and every button also play with the pointer,
# touch, or Tab and Enter (the gamepad's D-pad moves focus, South presses).
engine.group.slot1=Care actions
control.slot1=Digit1+Numpad1 | Action 1
control.slot2=Digit2+Numpad2 | Action 2
control.slot3=Digit3+Numpad3 | Action 3
control.slot4=Digit4+Numpad4 | Action 4
control.slot5=Digit5+Numpad5 | Action 5
control.slot6=Digit6+Numpad6 | Action 6
control.slot7=Digit7+Numpad7 | Action 7
control.slot8=Digit8+Numpad8 | Action 8
control.slot9=Digit9+Numpad9 | Action 9
control.slot10=Digit0+Numpad0 | Action 10
engine.group.hint=Session
control.hint=KeyH | Hint
control.undo=KeyU | Undo
control.camera=KeyC | Camera
control.pause=Escape | Pause

# Gamepad defaults: Start pauses (or closes a sheet), Y hints, X undoes,
# Select recentres the camera; the tray is reached by D-pad focus and South.
engine.gamepad.hint=North
engine.gamepad.undo=West
engine.gamepad.camera=Select
engine.gamepad.pause=Start
